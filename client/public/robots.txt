# `Sitemap` is a global directive: it applies regardless of position or of which
# user-agent group precedes it.

# ─── Search engines: everything that is worth indexing ──────────────────────────
User-agent: *
Content-Signal: search=yes,ai-train=no,use=reference
Allow: /
# The cue library and the music: 270-odd files, several megabytes, and nothing a
# search result could ever point at. Crawling them is pure bandwidth.
Disallow: /sounds/
Disallow: /music/
# The sprites. They are the game's art, not illustrations of it — a lone robot
# hull on a transparent background is worth nothing in image search, and there are
# about fifty of them. `social-card.jpg` and the icons are deliberately left out
# of this: those are what an unfurler and a browser tab actually ask for.
Disallow: /*.webp$
# The update check's answer, emitted by the build. A machine-readable file about
# which deploy is current, with no reader but the game itself.
Disallow: /version.json
#
# Deliberately still crawlable: `/assets/*`. It looks like the biggest thing to
# shut out, but it is the JS and CSS bundles, and Googlebot renders the page
# before indexing it — block them and the crawler sees the boot placeholder
# instead of the game. `/` and `/news/` are what we want indexed, and both are in
# the sitemap; neither needs its own `Allow`, since `Allow: /` above covers them.
